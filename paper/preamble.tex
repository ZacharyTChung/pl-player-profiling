% Shared preamble for the article and its supplement, so the two documents cannot drift
% apart in font, palette, float behaviour or number formatting.
\PassOptionsToPackage{hyphens}{url}

\usepackage[margin=1in]{geometry}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{booktabs}
\usepackage{graphicx}
\usepackage{siunitx}
\usepackage{subcaption}
% Citation style, and the one thing to change when the target venue changes. Numeric and
% compressed is the default, because a run of three references prints as [1-3] rather than
% as three author-year parentheticals, which is what keeps a citation-dense paragraph
% readable at this length. JQAS asks for Harvard author-date instead: comment the first
% line and the \bibliographystyle below, and uncomment their partners. Both compile; every
% citation in the sources is a natbib \citep or \citet, which works either way.
\usepackage[numbers,sort&compress]{natbib}
%\usepackage{natbib}                       % Harvard author-date
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{longtable}
\usepackage{microtype}
\usepackage[hidelinks]{hyperref}
% Cross-document references. The article and the supplement each point at the other, so
% both are compiled twice in sequence and each reads the other's .aux file.
\usepackage{xr-hyper}
% Keep every float inside the section that refers to it. A barrier at subsection level
% was tried and is too strong at this length: it gives every figure a page of its own,
% because a figure declared near the end of a subsection cannot flow into the next one.
\usepackage[section]{placeins}
\usepackage{etoolbox}

\bibliographystyle{unsrtnat}
%\bibliographystyle{plainnat}              % Harvard author-date
\graphicspath{{figures/}}

\sisetup{
  detect-all,
  group-separator = {,},
  round-mode = places,
  round-precision = 3,
}

\setlength{\parskip}{0pt}
\setlength{\emergencystretch}{3em}

% Both documents are figure dense. With LaTeX's defaults the float queue cannot drain
% fast enough and figures drift many pages past the text that refers to them.
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.75}

% --- Submission mode -----------------------------------------------------------------
% Compile with `latexmk -pdf -usepretex='\def\reviewmode{}' main.tex`, or `make review`,
% to get the manuscript most journals in this area ask for at submission: double spaced,
% line numbered, and with the authors and the repository URL removed so the review can be
% blind. Without the flag the same source gives the typeset article.
\newif\ifreview
\ifdefined\reviewmode\reviewtrue\fi

\ifreview
  \usepackage{setspace}
  \doublespacing
  \usepackage{lineno}
  \linenumbers
\fi

% The repository identifies the authors, so under review it is withheld rather than
% removed: a reviewer should still know the work is reproducible.
\ifreview
  \newcommand{\repourl}{[repository URL withheld for blind review]}
\else
  \newcommand{\repourl}{\url{https://github.com/ZacharyTChung/pl-player-profiling}}
\fi
